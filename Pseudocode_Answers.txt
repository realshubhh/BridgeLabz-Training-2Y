--------------------------------------------------------------------
Problem PC-1: Prime Number Checker
--------------------------------------------------------------------
INPUT : N (a positive integer, N > 1)
OUTPUT: "Prime" or "Not Prime"

ALGORITHM IsPrime(N):
    1.  IF N == 2 THEN
    2.      RETURN "Prime"
    3.  END IF
    4.  IF N MOD 2 == 0 THEN
    5.      RETURN "Not Prime"          // eliminate even numbers early
    6.  END IF
    7.  SET i = 3
    8.  WHILE i * i <= N DO             // only test factors up to sqrt(N)
    9.      IF N MOD i == 0 THEN
    10.         RETURN "Not Prime"
    11.     END IF
    12.     i = i + 2                    // skip even divisors
    13. END WHILE
    14. RETURN "Prime"
END ALGORITHM

Trace  - Example 1: N = 61
    - N is odd, so proceed.
    - i = 3, i*i = 9 <= 61, 61 MOD 3 != 0
    - i = 5, i*i = 25 <= 61, 61 MOD 5 != 0
    - i = 7, i*i = 49 <= 61, 61 MOD 7 != 0
    - i = 9, i*i = 81 > 61  -> loop ends
    - OUTPUT: Prime

Trace - Example 2: N = 8
    - N MOD 2 == 0 -> immediately
    - OUTPUT: Not Prime


--------------------------------------------------------------------
Problem PC-2: Number Digit Reversal
--------------------------------------------------------------------
INPUT : N (an integer)
OUTPUT: Reversed integer

ALGORITHM ReverseDigits(N):
    1.  SET reversed = 0
    2.  SET isNegative = FALSE
    3.  IF N < 0 THEN
    4.      isNegative = TRUE
    5.      N = -N                       // work with absolute value
    6.  END IF
    7.  WHILE N > 0 DO
    8.      SET digit = N MOD 10         // extract last digit
    9.      reversed = (reversed * 10) + digit   // append digit
    10.     N = N DIV 10                 // integer division, drop last digit
    11. END WHILE
    12. IF isNegative == TRUE THEN
    13.     reversed = -reversed
    14. END IF
    15. RETURN reversed
END ALGORITHM

Trace - Example 1: N = 1234
    N=1234 -> digit=4, reversed=4,    N=123
    N=123  -> digit=3, reversed=43,   N=12
    N=12   -> digit=2, reversed=432,  N=1
    N=1    -> digit=1, reversed=4321, N=0  -> loop ends
    OUTPUT: 4321

Trace - Example 2: N = 806
    N=806 -> digit=6, reversed=6,   N=80
    N=80  -> digit=0, reversed=60,  N=8
    N=8   -> digit=8, reversed=608, N=0  -> loop ends
    OUTPUT: 608
